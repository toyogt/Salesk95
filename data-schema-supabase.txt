orders_non_buyers_trigger_func()
DECLARE
    v_outlet_id TEXT;
    v_order_month DATE;
BEGIN
    IF TG_OP = 'DELETE' THEN
        v_outlet_id := OLD.outlet_id;
        v_order_month := date_trunc('month', OLD.created_at)::DATE;
    ELSE
        v_outlet_id := NEW.outlet_id;
        v_order_month := date_trunc('month', NEW.created_at)::DATE;
    END IF;

    PERFORM update_non_buyers_for_outlet_month(v_outlet_id, v_order_month);

    IF TG_OP = 'DELETE' THEN
        RETURN OLD;
    ELSE
        RETURN NEW;
    END IF;
END;

BEGIN
    SELECT order_number INTO order_number_text FROM orders WHERE id = NEW.id;

    effective_dist_id := COALESCE(NEW.fulfilled_by_distributor_id, NEW.distributor_id);

    IF (OLD.order_status = 'Draft'
        AND NEW.order_status IN ('Pending', 'Delivered')
        AND NEW.fulfillment_type = 'Distributor')
    THEN
        IF OLD.inventory_deducted = TRUE THEN
            RAISE NOTICE 'Inventory already deducted for order %', NEW.id;
            RETURN NEW;
        END IF;

        FOR order_item IN
            SELECT oi.product_id, oi.qty, p.product_name
            FROM order_items oi
            JOIN products p ON p.product_id = oi.product_id
            WHERE oi.order_id = NEW.id
        LOOP
            SELECT quantity_on_hand INTO current_stock
            FROM distributor_inventory
            WHERE distributor_id = effective_dist_id
              AND product_id = order_item.product_id
            FOR UPDATE;

            IF current_stock IS NULL THEN
                INSERT INTO distributor_inventory (distributor_id, product_id, quantity_on_hand, reorder_level)
                VALUES (effective_dist_id, order_item.product_id, 0, 5)
                ON CONFLICT (distributor_id, product_id) DO NOTHING;
                current_stock := 0;
            END IF;

            IF current_stock >= order_item.qty THEN
                UPDATE distributor_inventory
                SET quantity_on_hand = quantity_on_hand - order_item.qty,
                    updated_at = NOW()
                WHERE distributor_id = effective_dist_id
                  AND product_id = order_item.product_id
                RETURNING quantity_on_hand INTO new_balance;

                previous_balance := current_stock;

                INSERT INTO inventory_transactions (
                    distributor_id, product_id, transaction_type, quantity_change,
                    previous_balance, new_balance, created_by_email, order_id, reference_note
                ) VALUES (
                    effective_dist_id, order_item.product_id, 'SALE', -order_item.qty,
                    previous_balance, new_balance, NEW.created_by_email, NEW.id,
                    format('Order moved to %s from Draft: #%s - %s',
                           NEW.order_status, order_number_text, order_item.product_name)
                );
            ELSE
                RAISE EXCEPTION
                    'Insufficient stock for product % (distributor %): available %, need %',
                    order_item.product_id, effective_dist_id, current_stock, order_item.qty;
            END IF;
        END LOOP;

        NEW.inventory_deducted := TRUE;

    ELSIF (OLD.order_status IN ('Pending', 'Delivered')
           AND NEW.order_status IN ('Draft', 'Cancelled')
           AND OLD.fulfillment_type = 'Distributor')
    THEN
        IF OLD.inventory_deducted = TRUE THEN
            effective_dist_id := COALESCE(OLD.fulfilled_by_distributor_id, OLD.distributor_id);

            FOR order_item IN
                SELECT oi.product_id, oi.qty, p.product_name
                FROM order_items oi
                JOIN products p ON p.product_id = oi.product_id
                WHERE oi.order_id = NEW.id
            LOOP
                SELECT quantity_on_hand INTO previous_balance
                FROM distributor_inventory
                WHERE distributor_id = effective_dist_id
                  AND product_id = order_item.product_id;

                previous_balance := COALESCE(previous_balance, 0);

                UPDATE distributor_inventory
                SET quantity_on_hand = quantity_on_hand + order_item.qty,
                    updated_at = NOW()
                WHERE distributor_id = effective_dist_id
                  AND product_id = order_item.product_id
                RETURNING quantity_on_hand INTO new_balance;

                INSERT INTO inventory_transactions (
                    distributor_id, product_id, transaction_type, quantity_change,
                    previous_balance, new_balance, created_by_email, order_id, reference_note
                ) VALUES (
                    effective_dist_id, order_item.product_id, 'RETURN_IN', order_item.qty,
                    previous_balance, new_balance, NEW.created_by_email, NEW.id,
                    format('Order moved to %s from %s: #%s - %s',
                           NEW.order_status, OLD.order_status, order_number_text, order_item.product_name)
                );
            END LOOP;

            NEW.inventory_deducted := FALSE;
        END IF;

    ELSIF (OLD.order_status = 'Pending'
           AND NEW.order_status = 'Delivered'
           AND NEW.fulfillment_type = 'Distributor'
           AND OLD.inventory_deducted = TRUE)
    THEN
        effective_dist_id := COALESCE(NEW.fulfilled_by_distributor_id, NEW.distributor_id);

        FOR order_item IN
            SELECT oi.product_id, oi.qty, p.product_name
            FROM order_items oi
            JOIN products p ON p.product_id = oi.product_id
            WHERE oi.order_id = NEW.id
        LOOP
            SELECT quantity_on_hand INTO current_stock
            FROM distributor_inventory
            WHERE distributor_id = effective_dist_id
              AND product_id = order_item.product_id;

            INSERT INTO inventory_transactions (
                distributor_id, product_id, transaction_type, quantity_change,
                previous_balance, new_balance, created_by_email, order_id, reference_note
            ) VALUES (
                effective_dist_id, order_item.product_id, 'DELIVERY_NOTE', 0,
                current_stock, current_stock, NEW.created_by_email, NEW.id,
                format('Order marked Delivered from Pending: #%s - %s',
                       order_number_text, order_item.product_name)
            );
        END LOOP;
    END IF;

    RETURN NEW;
END;

handle_visit_order_created()
BEGIN
    UPDATE public.visits
    SET order_created = true
    WHERE outlet_id = NEW.outlet_id
      AND visit_date = NEW.created_at::date
      AND order_created = false;
    RETURN NEW;
END;

set_visit_order_created()

BEGIN
    -- Check if any order (non-Draft, non-Cancelled) exists for this outlet & date
    IF EXISTS (
        SELECT 1 FROM public.orders o
        WHERE o.outlet_id = NEW.outlet_id
          AND o.created_at::date = NEW.visit_date
          AND o.order_status NOT IN ('Draft', 'Cancelled')  -- adjust as needed
    ) THEN
        NEW.order_created = true;
    ELSE
        NEW.order_created = false;   -- explicitly set false if no order
    END IF;
    RETURN NEW;
END;

check_inventory()

DECLARE
    available_stock INTEGER;
BEGIN
    SELECT quantity_on_hand INTO available_stock
    FROM distributor_inventory
    WHERE distributor_id = p_distributor_id
    AND product_id = p_product_id;
    
    IF available_stock IS NULL THEN
        RETURN jsonb_build_object(
            'available', false,
            'stock', 0,
            'required', p_required_qty,
            'message', 'No stock record found for this product'
        );
    ELSIF available_stock >= p_required_qty THEN
        RETURN jsonb_build_object(
            'available', true,
            'stock', available_stock,
            'required', p_required_qty,
            'message', 'Sufficient stock available'
        );
    ELSE
        RETURN jsonb_build_object(
            'available', false,
            'stock', available_stock,
            'required', p_required_qty,
            'message', format('Insufficient stock. Available: %s, Required: %s', available_stock, p_required_qty)
        );
    END IF;
END;
